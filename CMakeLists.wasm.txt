# PRIORITY 1: Enable WASM SIMD optimizations
# ===========================================
set(WEBP_ENABLE_SIMD ON CACHE BOOL "Enable WASM SIMD optimizations" FORCE)
message(STATUS "WASM SIMD optimizations enabled")

# PRIORITY 1: Emscripten-Specific Compiler Flags
# ==============================================

# Core WASM flags with maximum modern browser features
set(WASM_CORE_FLAGS
    -s WASM=1
    -s MODULARIZE=1
    -s EXPORT_ES6=1
    -O3
    -flto                         # Link-time optimization for maximum performance
    --closure=1                   # Closure compiler for size optimization
    -s ALLOW_MEMORY_GROWTH=1
    -s "EXPORTED_RUNTIME_METHODS=['ccall','cwrap','FS','setValue','getValue']"
)

# Maximum WASM SIMD and modern features
if(WEBP_ENABLE_SIMD)
  list(APPEND WASM_CORE_FLAGS
    -msimd128                     # WASM SIMD instructions
    -s SIMD=1                     # Enable SIMD support
    -msse2                        # Additional SIMD optimizations via WASM SIMD
    -msse3
    -mssse3
  )
endif()

# Aggressive memory configuration for native-level performance
set(WASM_MEMORY_FLAGS
    -s INITIAL_MEMORY=67108864    # 64MB initial (larger for image processing)
    -s MAXIMUM_MEMORY=2147483648  # 2GB maximum (no artificial browser limits)
    -s STACK_SIZE=2097152         # 2MB stack for complex operations
    -s TOTAL_STACK=2097152        # Match stack allocation
)

# Advanced browser features and filesystem support
set(WASM_ADVANCED_FLAGS
    -s FORCE_FILESYSTEM=1
    -s "EXTRA_EXPORTED_RUNTIME_METHODS=['FS','IDBFS','writeFile','readFile']"
    # Modern browser APIs (require WebGPU+SIMD browsers only)
    -s USE_PTHREADS=0             # Disable for SIDE_MODULE compatibility, enable in MAIN_MODULE
    -s SHARED_MEMORY=0            # Disable for SIDE_MODULE, enable in MAIN_MODULE
    -s ASYNCIFY=1                 # Enable async/await in C code
    -s ASYNCIFY_STACK_SIZE=8192   # Stack for async operations
    # Advanced Emscripten features
    -s FETCH=1                    # Modern fetch API for network operations
    -s WASM_BIGINT=1             # BigInt support for large numbers
    -s DYNAMIC_EXECUTION=0        # Security: disable eval-like features
    -s TEXTDECODER=2             # Fast text decoding
    # Eliminate unnecessary runtime methods for smaller size
    -s MINIMAL_RUNTIME=0          # Keep full runtime for flexibility
)

# Additional flags for MAIN_MODULE builds (threading-capable)
set(WASM_MAIN_MODULE_FLAGS
    -s USE_PTHREADS=1             # Enable threading for MAIN_MODULE
    -s SHARED_MEMORY=1            # Enable SharedArrayBuffer
    -s PTHREAD_POOL_SIZE=4        # Thread pool for parallel operations
    -s PROXY_TO_PTHREAD=1         # Keep main thread responsive
)

# Build type detection and flag combination
option(BUILD_SIDE_MODULE "Build as SIDE_MODULE for dynamic loading" OFF)
option(BUILD_MAIN_MODULE "Build as MAIN_MODULE with threading" ON)

# Combine flags based on build type
if(BUILD_SIDE_MODULE)
    # SIDE_MODULE: No threading, optimized for dlopen()
    set(WASM_FLAGS ${WASM_CORE_FLAGS} ${WASM_MEMORY_FLAGS} ${WASM_ADVANCED_FLAGS})
    list(APPEND WASM_FLAGS -s SIDE_MODULE=1)
    list(APPEND WASM_FLAGS -s ERROR_ON_UNDEFINED_SYMBOLS=0)
    message(STATUS "Building libwebp.wasm as SIDE_MODULE")
elseif(BUILD_MAIN_MODULE)
    # MAIN_MODULE: Full features including threading
    set(WASM_FLAGS ${WASM_CORE_FLAGS} ${WASM_MEMORY_FLAGS} ${WASM_ADVANCED_FLAGS} ${WASM_MAIN_MODULE_FLAGS})
    list(APPEND WASM_FLAGS -s MAIN_MODULE=1)
    message(STATUS "Building libwebp.wasm as MAIN_MODULE with threading")
else()
    # Default standalone build
    set(WASM_FLAGS ${WASM_CORE_FLAGS} ${WASM_MEMORY_FLAGS} ${WASM_ADVANCED_FLAGS})
    message(STATUS "Building libwebp.wasm as standalone module")
endif()

# Apply WASM flags to C and CXX compilers
string(REPLACE ";" " " WASM_FLAGS_STR "${WASM_FLAGS}")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${WASM_FLAGS_STR}")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${WASM_FLAGS_STR}")

# Linker flags for WASM
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} ${WASM_FLAGS_STR}")
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} ${WASM_FLAGS_STR}")

# PRIORITY 1: Library Creation
# ============================

if(WEBP_BUILD_STATIC)
  # Create static library
  add_library(webp STATIC ${WEBP_SRCS})

  # Include directories
  target_include_directories(webp
                             PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>"
                             PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>"
                             INTERFACE "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")

  # Set target properties
  set_target_properties(webp PROPERTIES
                        OUTPUT_NAME "webp"
                        VERSION "1.4.0"
                        SOVERSION "1"
                        POSITION_INDEPENDENT_CODE TRUE)

  # WASM-specific compile definitions
  target_compile_definitions(webp PUBLIC
    WEBP_HAVE_WASM_SIMD=1
    WEBP_WASM_NATIVE=1
  )

  # Add SIMD support definitions
  if(WEBP_ENABLE_SIMD)
    target_compile_definitions(webp PUBLIC
      WEBP_HAVE_SIMD=1
      WEBP_USE_SIMD=1
    )
  endif()

  # Set as main WebP target
  set(WEBP_LIBRARY_TARGET webp)

  message(STATUS "Created static libwebp.wasm target: webp")
endif()

# PRIORITY 2: Installation Configuration
# ======================================

if(WEBP_BUILD_STATIC AND NOT SKIP_INSTALL_LIBRARIES AND NOT SKIP_INSTALL_ALL)
  install(TARGETS ${WEBP_LIBRARY_TARGET}
          EXPORT libwebp-wasm
          ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
endif()

# Install headers
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/src/webp"
        DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
        FILES_MATCHING PATTERN "*.h")

# PRIORITY 1: Export Configuration
# ================================

if(WEBP_BUILD_STATIC)
  install(EXPORT libwebp-wasm

          DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libwebp-wasm"
          NAMESPACE libwebp-wasm::)

  # Create version file
  include(CMakePackageConfigHelpers)
  write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/libwebp-wasm-config-version.cmake"
    VERSION "1.4.0"
    COMPATIBILITY SameMajorVersion
  )

  install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/libwebp-wasm-config-version.cmake"
